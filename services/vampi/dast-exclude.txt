# URLs the DAST lane keeps out of the scan (scripts/dast-lane.mjs): one regular expression per line.
# GET /createdb drops and reseeds the database (VN-05). Scanned, it resets the tables while other
# operations are being attacked, and their answers turn into "no such table" 500s (VN-01, VN-02).
^https?://[^/]+/createdb.*$
# GET /users/v1/_debug answers every user's password (VN-06): kept out so ZAP never holds them.
^https?://[^/]+/users/v1/_debug.*$
# PUT /users/v1/{username}/password changes the password of the token's own user in secure mode,
# whatever the path names: an authenticated scan would change its own credentials and lose them
# at the next renewal. Without a token it answers 401 before its handler runs, so leaving it out
# costs the unauthenticated scan nothing.
^https?://[^/]+/users/v1/[^/]+/password$
